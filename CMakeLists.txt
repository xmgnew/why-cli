cmake_minimum_required(VERSION 3.16)
project(why VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(WHY_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(why_core src/model.c src/procfs.c src/metadata.c src/history.c src/cpu_analysis.c src/attribution.c)
target_include_directories(why_core PUBLIC src)
target_compile_definitions(why_core PUBLIC _POSIX_C_SOURCE=200809L)
if(APPLE)
  target_compile_definitions(why_core PUBLIC _DARWIN_C_SOURCE)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(why_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(WHY_SANITIZE)
    target_compile_options(why_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(why_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(why src/main.c)
target_link_libraries(why PRIVATE why_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(why_tests tests/test_core.c)
  target_link_libraries(why_tests PRIVATE why_core)
  add_test(NAME core COMMAND why_tests)
  add_executable(why_history_tests tests/test_history.c)
  target_link_libraries(why_history_tests PRIVATE why_core)
  add_test(NAME history COMMAND why_history_tests)
  add_executable(why_cpu_tests tests/test_cpu_analysis.c)
  target_link_libraries(why_cpu_tests PRIVATE why_core)
  add_test(NAME cpu_analysis COMMAND why_cpu_tests)
  add_executable(why_attribution_tests tests/test_attribution.c)
  target_link_libraries(why_attribution_tests PRIVATE why_core)
  add_test(NAME attribution COMMAND why_attribution_tests)
  add_test(NAME cli_help COMMAND why --help)
  add_test(NAME cli_invalid_count COMMAND why sample --count -1)
  set_tests_properties(cli_invalid_count PROPERTIES WILL_FAIL TRUE)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_test(NAME linux_lifecycle
      COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/linux_lifecycle.sh $<TARGET_FILE:why>)
    set_tests_properties(linux_lifecycle PROPERTIES TIMEOUT 30)
  endif()
endif()
install(TARGETS why RUNTIME DESTINATION bin)
